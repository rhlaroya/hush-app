<!DOCTYPE html>

<html lang="en">

<head>

  <meta charset="UTF-8">

  <meta name="viewport" content="width=device-width, initial-scale=1.0">

  <title>Hush</title>

  <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>

  <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

  <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

  <script src="https://cdn.jsdelivr.net/npm/@emailjs/browser@3/dist/email.min.js"></script>

  <script src="https://cdn.tailwindcss.com"></script>

</head>

<body>

  <div id="root"></div>


  <script type="text/babel">

    const { useState, useEffect, useRef } = React;


    function HushApp() {

      const [nickname, setNickname] = useState('');

      const [currentUser, setCurrentUser] = useState(null);

      const [messages, setMessages] = useState([]);

      const [inputMessage, setInputMessage] = useState('');

      const [error, setError] = useState('');

      const [onlineUsers, setOnlineUsers] = useState({});

      const messagesEndRef = useRef(null);


      // EmailJS Configuration

      const EMAILJS_SERVICE_ID = 'service_537plwx';

      const EMAILJS_TEMPLATE_ID = 'template_dc63hon';

      const EMAILJS_PUBLIC_KEY = '_nbn7K4x0VrDxW0UI';


      // Display names mapping

      const getDisplayName = (user) => {

        return user === 'regae' ? 'heirofvenus' : 'heiressofvenus';

      };


      // Load user session and initialize EmailJS

      useEffect(() => {

        const stored = sessionStorage.getItem('hushvenus_user');

        if (stored) {

          setCurrentUser(stored.toLowerCase());

        }

        

        // Initialize EmailJS

        emailjs.init(EMAILJS_PUBLIC_KEY);

        console.log('EmailJS initialized successfully');

      }, []);


      // Online status management

      useEffect(() => {

        if (!currentUser) return;


        const markOnline = () => {

          const status = {

            user: currentUser,

            timestamp: Date.now()

          };

          sessionStorage.setItem('hushvenus_status_' + currentUser, JSON.stringify(status));

        };


        markOnline();

        const interval = setInterval(markOnline, 2000);


        const checkStatus = () => {

          const otherUser = currentUser === 'regae' ? 'phish' : 'regae';

          const statusStr = sessionStorage.getItem('hushvenus_status_' + otherUser);

          

          const newOnlineUsers = {};

          if (statusStr) {

            const status = JSON.parse(statusStr);

            if (Date.now() - status.timestamp < 5000) {

              newOnlineUsers[otherUser] = true;

            }

          }

          setOnlineUsers(newOnlineUsers);

        };


        checkStatus();

        const statusInterval = setInterval(checkStatus, 2000);


        return () => {

          clearInterval(interval);

          clearInterval(statusInterval);

          sessionStorage.removeItem('hushvenus_status_' + currentUser);

        };

      }, [currentUser]);


      // Auto-scroll to bottom

      useEffect(() => {

        messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });

      }, [messages]);


      // Handle login

      const handleLogin = () => {

        const normalized = nickname.trim().toLowerCase();

        

        if (normalized === 'regae' || normalized === 'phish') {

          setCurrentUser(normalized);

          sessionStorage.setItem('hushvenus_user', normalized);

          setError('');

        } else {

          setError('Sorry, that\'s not the right nickname.');

        }

      };


      const handleLoginKeyPress = (e) => {

        if (e.key === 'Enter') {

          handleLogin();

        }

      };


      // Send message

      const sendMessage = () => {

        if (!inputMessage.trim()) return;


        const newMessage = {

          id: Date.now(),

          text: inputMessage,

          sender: currentUser,

          timestamp: Date.now()

        };


        setMessages(prev => [...prev, newMessage]);

        setInputMessage('');


        setTimeout(() => {

          setMessages(prev => prev.filter(msg => msg.id !== newMessage.id));

        }, 3000);

      };


      const handleMessageKeyPress = (e) => {

        if (e.key === 'Enter' && !e.shiftKey) {

          e.preventDefault();

          sendMessage();

        }

      };


      // Ping other user

      const pingUser = async () => {

        const otherUser = currentUser === 'regae' ? 'phish' : 'regae';

        const otherDisplayName = getDisplayName(otherUser);

        const email = currentUser === 'regae' 

          ? 'sophiapecina24@yahoo.com' 

          : 'rhlaroya@gmail.com';


        try {

          const result = await emailjs.send(

            EMAILJS_SERVICE_ID,

            EMAILJS_TEMPLATE_ID,

            {

              to_email: email,

              to_name: otherDisplayName,

              from_name: getDisplayName(currentUser),

              message: 'talk?',

              subject: 'talk?'

            }

          );


          console.log('EmailJS Success:', result);

          

          const notification = {

            id: Date.now(),

            text: `📧 Ping sent to ${otherDisplayName}`,

            sender: 'system',

            timestamp: Date.now()

          };

          setMessages(prev => [...prev, notification]);

          

          setTimeout(() => {

            setMessages(prev => prev.filter(msg => msg.id !== notification.id));

          }, 3000);


        } catch (err) {

          console.error('Ping error:', err);

          const notification = {

            id: Date.now(),

            text: `❌ Error: ${err.text || err.message || 'Failed to send'}`,

            sender: 'system',

            timestamp: Date.now()

          };

          setMessages(prev => [...prev, notification]);

          

          setTimeout(() => {

            setMessages(prev => prev.filter(msg => msg.id !== notification.id));

          }, 3000);

        }

      };


      // Logout

      const logout = () => {

        sessionStorage.removeItem('hushvenus_user');

        setCurrentUser(null);

        setMessages([]);

      };


      // Login page

      if (!currentUser) {

        return (

          <div className="min-h-screen bg-gradient-to-br from-purple-900 via-pink-900 to-red-900 flex items-center justify-center p-4">

            <div className="bg-white/10 backdrop-blur-lg rounded-3xl p-8 max-w-md w-full shadow-2xl">

              <h1 className="text-4xl font-bold text-white text-center mb-2">Hush</h1>

              <p className="text-pink-200 text-center mb-8">Messages here never existed nor happened</p>

              

              <div className="space-y-4">

                <div>

                  <label className="block text-white mb-2 text-sm font-medium">

                    What is the nickname I gave you?

                  </label>

                  <input

                    type="text"

                    value={nickname}

                    onChange={(e) => setNickname(e.target.value)}

                    onKeyPress={handleLoginKeyPress}

                    className="w-full px-4 py-3 rounded-lg bg-white/20 text-white placeholder-pink-200 border border-white/30 focus:outline-none focus:ring-2 focus:ring-pink-400"

                    placeholder="Enter your nickname"

                    autoFocus

                  />

                </div>

                

                {error && (

                  <p className="text-red-300 text-sm">{error}</p>

                )}

                

                <button

                  onClick={handleLogin}

                  className="w-full bg-gradient-to-r from-pink-500 to-purple-500 text-white py-3 rounded-lg font-semibold hover:from-pink-600 hover:to-purple-600 transition-all transform hover:scale-105"

                >

                  Enter

                </button>

              </div>

            </div>

          </div>

        );

      }


      // Chat page

      return (

        <div className="min-h-screen bg-gradient-to-br from-purple-900 via-pink-900 to-red-900 flex flex-col">

          <div className="bg-white/10 backdrop-blur-lg border-b border-white/20 p-4">

            <div className="max-w-4xl mx-auto flex items-center justify-between">

              <div>

                <h1 className="text-2xl font-bold text-white">Hush</h1>

                <p className="text-pink-200 text-sm">Logged in as {getDisplayName(currentUser)}</p>

              </div>

              <div className="flex items-center gap-3">

                <div className="flex items-center gap-2 bg-white/10 px-3 py-2 rounded-lg">

                  <div className={`w-2 h-2 rounded-full ${onlineUsers[currentUser === 'regae' ? 'phish' : 'regae'] ? 'bg-green-400' : 'bg-gray-400'}`}></div>

                  <span className="text-white text-sm">

                    {getDisplayName(currentUser === 'regae' ? 'phish' : 'regae')} - {onlineUsers[currentUser === 'regae' ? 'phish' : 'regae'] ? 'Online' : 'Offline'}

                  </span>

                </div>

                

                <button

                  onClick={pingUser}

                  className="flex items-center gap-2 bg-yellow-500 text-white px-4 py-2 rounded-lg font-medium hover:bg-yellow-600 transition-all"

                >

                  <svg width="18" height="18" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">

                    <path d="M18 8A6 6 0 0 0 6 8c0 7-3 9-3 9h18s-3-2-3-9"></path>

                    <path d="M13.73 21a2 2 0 0 1-3.46 0"></path>

                  </svg>

                  Ping {getDisplayName(currentUser === 'regae' ? 'phish' : 'regae')}

                </button>

                <button

                  onClick={logout}

                  className="bg-white/20 text-white px-4 py-2 rounded-lg font-medium hover:bg-white/30 transition-all"

                >

                  Logout

                </button>

              </div>

            </div>

          </div>


          <div className="flex-1 overflow-y-auto p-4">

            <div className="max-w-4xl mx-auto space-y-3">

              {messages.length === 0 && (

                <div className="text-center text-pink-200 mt-20">

                  <p className="text-lg">No messages yet. Start chatting!</p>

                  <p className="text-sm mt-2">Messages here never existed nor happened ✨</p>

                </div>

              )}

              

              {messages.map((msg) => (

                <div

                  key={msg.id}

                  className={`flex ${msg.sender === 'system' ? 'justify-center' : msg.sender === currentUser ? 'justify-end' : 'justify-start'} animate-fade-in`}

                >

                  <div

                    className={`max-w-xs px-4 py-2 rounded-2xl ${

                      msg.sender === 'system'

                        ? 'bg-white/20 text-white text-sm'

                        : msg.sender === currentUser

                        ? 'bg-gradient-to-r from-pink-500 to-purple-500 text-white'

                        : 'bg-white/90 text-gray-800'

                    }`}

                  >

                    {msg.sender !== 'system' && (

                      <p className="text-xs opacity-70 mb-1">

                        {getDisplayName(msg.sender)}

                      </p>

                    )}

                    <p>{msg.text}</p>

                  </div>

                </div>

              ))}

              <div ref={messagesEndRef} />

            </div>

          </div>


          <div className="bg-white/10 backdrop-blur-lg border-t border-white/20 p-4">

            <div className="max-w-4xl mx-auto flex gap-2">

              <input

                type="text"

                value={inputMessage}

                onChange={(e) => setInputMessage(e.target.value)}

                onKeyPress={handleMessageKeyPress}

                placeholder="Type a message..."

                className="flex-1 px-4 py-3 rounded-lg bg-white/20 text-white placeholder-pink-200 border border-white/30 focus:outline-none focus:ring-2 focus:ring-pink-400"

              />

              <button

                onClick={sendMessage}

                className="bg-gradient-to-r from-pink-500 to-purple-500 text-white px-6 py-3 rounded-lg font-semibold hover:from-pink-600 hover:to-purple-600 transition-all transform hover:scale-105 flex items-center gap-2"

              >

                <svg width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2">

                  <line x1="22" y1="2" x2="11" y2="13"></line>

                  <polygon points="22 2 15 22 11 13 2 9 22 2"></polygon>

                </svg>

                Send

              </button>

            </div>

          </div>


          <style>{`

            @keyframes fade-in {

              from {

                opacity: 0;

                transform: translateY(10px);

              }

              to {

                opacity: 1;

                transform: translateY(0);

              }

            }

            .animate-fade-in {

              animation: fade-in 0.3s ease-out;

            }

          `}</style>

        </div>

      );

    }


    ReactDOM.render(<HushApp />, document.getElementById('root'));

  </script>

</body>

</html>
